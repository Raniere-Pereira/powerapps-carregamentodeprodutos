scrCarregamento:
  tipo: Screen
  objetivo: Registrar volumes em uma NF, validar regra do próximo volume e atualizar status da NF.
  propriedades:
    Fill: =RGBA(234, 234, 234, 1)
    LoadingSpinnerColor: =RGBA(164, 38, 44, 1)
    OnVisible: |
      =Set(VarProcessando, false);
      Set(VarAba, "REGISTRO");
      If(
          !IsBlank(VarCabecalho),
          Refresh(expedicao_cabecalho);
          Refresh(Expedicao_Linhas);
          Set(VarCabecalho, LookUp(expedicao_cabecalho, ID = VarCabecalho.ID));
          ClearCollect(colLinhasNF, Filter(Expedicao_Linhas, Lin_CabecalhoID = VarCabecalho.ID && Lin_Status = "CARREGADO"));
          Set(VarVolumesCarregados, CountRows(colLinhasNF));
          Set(VarProximoVolume, Coalesce(First(Sort(Filter(Sequence(VarCabecalho.Volumes) As Vol, !(Vol.Value in colLinhasNF.Lin_NumeroVolume)), Value, SortOrder.Ascending)).Value, 0))
      )
  componentes:
    - nome: recInfo
      tipo: Rectangle@2.3.0
      propriedades:
        Fill: =RGBA(255, 255, 255, 1)
        Height: =185
        Width: =If(App.Width < 600, Parent.Width - 30, Max(320, (Parent.Width - 45) * 0.42))
        X: =15
        Y: =83
    - nome: lblTituloCarregamento
      tipo: Label@2.5.1
      propriedades:
        Text: ="Carregamento"
    - nome: lblNF
      tipo: Label@2.5.1
      propriedades:
        Text: |
          ="NF: " & VarCabecalho.NumeroNF
    - nome: lblCliente
      tipo: Label@2.5.1
      propriedades:
        Text: |
          ="Cliente: " & VarCabecalho.Cliente
    - nome: lblResumoVolume
      tipo: Label@2.5.1
      propriedades:
        Color: =RGBA(164, 38, 44, 1)
        Text: |
          ="Volumes carregados: " & VarVolumesCarregados & " / " & VarCabecalho.Volumes
    - nome: lbl_Saldos
      tipo: Label@2.5.1
      propriedades:
        Color: =RGBA(164, 38, 44, 1)
        Text: |
          ="Saldo: " & (VarCabecalho.Volumes - VarVolumesCarregados)
    - nome: drpTipoProduto
      tipo: Classic/ComboBox@2.4.0
      propriedades:
        DefaultSelectedItems: =["Máquina"]
        Items: =["Peça","Máquina"]
        OnChange: |
          =Reset(txtSerieProduto);
          Reset(txtPeca)
    - nome: txtSerieProduto
      tipo: Classic/TextInput@2.3.2
      propriedades:
        HintText: ="Digite o número de série"
        Visible: =drpTipoProduto.Selected.Value = "Máquina"
    - nome: txtPeca
      tipo: Classic/TextInput@2.3.2
      propriedades:
        HintText: ="Informe a peça/conteúdo do volume"
        Visible: =drpTipoProduto.Selected.Value = "Peça"
    - nome: lblProximoVolume
      tipo: Label@2.5.1
      propriedades:
        Text: |
          ="Próximo volume: " & If(VarProximoVolume > 0, Text(VarProximoVolume), "—")
    - nome: btnRegistrarVolume
      tipo: Classic/Button@2.2.0
      propriedades:
        Text: ="REGISTRAR VOLUME"
        OnSelect: |
          =If(
              !VarProcessando,
              Set(VarProcessando, true);
              Refresh(expedicao_cabecalho);
              Refresh(Expedicao_Linhas);
              Set(VarCabecalho, LookUp(expedicao_cabecalho, ID = VarCabecalho.ID));
              ClearCollect(colLinhasNF, Filter(Expedicao_Linhas, Lin_CabecalhoID = VarCabecalho.ID && Lin_Status = "CARREGADO"));
              Set(VarVolumesCarregados, CountRows(colLinhasNF));
              Set(VarProximoVolume, Coalesce(First(Sort(Filter(Sequence(VarCabecalho.Volumes) As Vol, !(Vol.Value in colLinhasNF.Lin_NumeroVolume)), Value, SortOrder.Ascending)).Value, 0));
              If(
                  VarCabecalho.Status.Value = "CONCLUÍDO" || VarCabecalho.Status.Value = "CANCELADO",
                  Notify("Esta NF não permite novos carregamentos.", NotificationType.Warning),
                  IsBlank(drpTipoProduto.Selected.Value),
                  Notify("Selecione o tipo de produto.", NotificationType.Warning),
                  drpTipoProduto.Selected.Value = "Máquina" && IsBlank(Trim(txtSerieProduto.Text)),
                  Notify("Informe o número de série da máquina.", NotificationType.Warning),
                  drpTipoProduto.Selected.Value = "Máquina" && !IsNumeric(Trim(txtSerieProduto.Text)),
                  Notify("A série deve conter apenas números.", NotificationType.Warning),
                  drpTipoProduto.Selected.Value = "Peça" && IsBlank(Trim(txtPeca.Text)),
                  Notify("Informe a peça.", NotificationType.Warning),
                  VarProximoVolume = 0 || VarVolumesCarregados >= VarCabecalho.Volumes,
                  Notify("Todos os volumes desta NF já foram carregados.", NotificationType.Warning),
                  drpTipoProduto.Selected.Value = "Máquina" && !IsBlank(LookUp(Expedicao_Linhas, Lin_SerieProduto = Value(Trim(txtSerieProduto.Text)) && Lin_Status = "CARREGADO")),
                  Notify("Esta série já possui um carregamento ativo.", NotificationType.Warning),
                  !IsBlank(LookUp(Expedicao_Linhas, Lin_ChaveVolume = Text(VarCabecalho.NumeroNF) & "-" & Text(VarProximoVolume) && Lin_Status = "CARREGADO")),
                  Notify("O volume que será registrado já possui um carregamento ativo.", NotificationType.Warning),
                  IfError(
                      Patch(Expedicao_Linhas, Defaults(Expedicao_Linhas), {Lin_CabecalhoID: VarCabecalho.ID, Lin_NumeroNF: VarCabecalho.NumeroNF, Lin_NumeroVolume: VarProximoVolume, Lin_ChaveVolume: Text(VarCabecalho.NumeroNF) & "-" & Text(VarProximoVolume), Lin_TipoProduto: {Value: drpTipoProduto.Selected.Value}, Lin_SerieProduto: If(drpTipoProduto.Selected.Value = "Máquina", Value(Trim(txtSerieProduto.Text)), Blank()), Lin_Peca: If(drpTipoProduto.Selected.Value = "Peça", Trim(txtPeca.Text), Blank()), Lin_DataCarregamento: Now(), Lin_UsuarioCarregamento: User().FullName, Lin_Status: "CARREGADO"}),
                      Notify("Erro ao registrar o volume: " & FirstError.Message, NotificationType.Error),
                      Patch(expedicao_cabecalho, LookUp(expedicao_cabecalho, ID = VarCabecalho.ID), {VolumesCarregados: VarVolumesCarregados + 1, Status: {Value: If(VarVolumesCarregados + 1 >= VarCabecalho.Volumes, "CONCLUÍDO", "EM CARREGAMENTO")}, DataCarregamento: If(VarVolumesCarregados + 1 >= VarCabecalho.Volumes, Now(), Blank())}),
                      Notify("Volume registrado, mas houve erro ao atualizar a NF: " & FirstError.Message, NotificationType.Error),
                      Notify("Volume " & Text(VarProximoVolume) & " registrado com sucesso.", NotificationType.Success);
                      Reset(txtSerieProduto);
                      Reset(txtPeca)
                  )
              );
              Refresh(expedicao_cabecalho);
              Refresh(Expedicao_Linhas);
              Set(VarCabecalho, LookUp(expedicao_cabecalho, ID = VarCabecalho.ID));
              ClearCollect(colLinhasNF, Filter(Expedicao_Linhas, Lin_CabecalhoID = VarCabecalho.ID && Lin_Status = "CARREGADO"));
              Set(VarVolumesCarregados, CountRows(colLinhasNF));
              Set(VarProximoVolume, Coalesce(First(Sort(Filter(Sequence(VarCabecalho.Volumes) As Vol, !(Vol.Value in colLinhasNF.Lin_NumeroVolume)), Value, SortOrder.Ascending)).Value, 0));
              Set(VarProcessando, false)
          )
    - nome: galCarregamento
      tipo: Gallery@2.15.0
      variante: BrowseLayout_Vertical_TwoTextOneImageVariant_ver5.0
      propriedades:
        Items: =Sort(colLinhasNF, Lin_NumeroVolume, SortOrder.Ascending)
        Visible: =App.Width >= 600 || VarAba = "VOLUMES"
      filhos:
        - nome: lblgalSeriePeca
          tipo: Label@2.5.1
          propriedades:
            Text: |
              =If(
                  ThisItem.Lin_TipoProduto.Value = "Máquina",
                  "Série: " & ThisItem.Lin_SerieProduto,
                  "Peça: " & ThisItem.Lin_Peca
              )
        - nome: lblgalVolume
          tipo: Label@2.5.1
          propriedades:
            Text: |
              ="Volume: " & ThisItem.Lin_NumeroVolume
    - nome: btnCarregamentoVoltarPendentes
      tipo: Classic/Button@2.2.0
      propriedades:
        Text: ="Pendentes"
        OnSelect: |
          =If(
              VarProcessando,
              Notify("Aguarde o registro do volume terminar.", NotificationType.Warning),
              Reset(txtSerieProduto);
              Reset(txtPeca);
              Navigate(scrPendentes, ScreenTransition.Fade)
          )
    - nome: btnCarregamentoVoltarConcluidas
      tipo: Classic/Button@2.2.0
      propriedades:
        Text: ="Concluídas"
        OnSelect: |
          =If(
              VarProcessando,
              Notify("Aguarde o registro do volume terminar.", NotificationType.Warning),
              Reset(txtSerieProduto);
              Reset(txtPeca);
              Navigate(scrConcluidas, ScreenTransition.Fade)
          )
    - nome: lblTituloRegistro
      tipo: Label@2.5.1
      propriedades:
        Text: ="Registro de Volume"
    - nome: lblTipoProduto
      tipo: Label@2.5.1
      propriedades:
        Text: ="Tipo de Produto*"
    - nome: lblSerieProduto
      tipo: Label@2.5.1
      propriedades:
        Text: ="Nº de Série*"
    - nome: lblTituloGalCarregamento
      tipo: Label@2.5.1
      propriedades:
        Text: ="Volumes carregados (" & VarVolumesCarregados & ")"
    - nome: btnAbaRegistro
      tipo: Classic/Button@2.2.0
      propriedades:
        Text: ="Registrar"
        OnSelect: =Set(VarAba, "REGISTRO")
    - nome: btnAbaVolumes
      tipo: Classic/Button@2.2.0
      propriedades:
        Text: ="Volumes (" & VarVolumesCarregados & ")"
        OnSelect: =Set(VarAba, "VOLUMES")
