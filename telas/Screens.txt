                      SortOrder.Descending
                  ),

                  IsNumeric(Trim(txtBuscaConcluidas.Text)),
                  Sort(
                      Filter(
                          expedicao_cabecalho,
                          Status.Value = "CONCLUÍDO" && NumeroNF = Value(Trim(txtBuscaConcluidas.Text))
                      ),
                      DataCarregamento,
                      SortOrder.Descending

                      Filter(
